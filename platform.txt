# ------------------------------------------------------------------------------
# 0. BUILD TOOLS PATHS
# ------------------------------------------------------------------------------
# Arduino connaît les chemins des outils grâce au fichier JSON
# (package_esp8266RTOS_index.json). On les récupère via {runtime.tools.<nom>.path}
# => pas besoin de coder les versions en dur ici.

cmake.path={runtime.tools.cmake.path}

python3.path={runtime.tools.python3.path}


# ------------------------------------------------------------------------------
# 1. PYTHON ENVIRONMENT CONFIGURATION
# ------------------------------------------------------------------------------
python.venv.path={runtime.platform.path}/env/bin/python
python.venv.path.windows={python3.path}/python.exe


# ------------------------------------------------------------------------------
# 2. PATH DEFINITIONS
# ------------------------------------------------------------------------------
compiler.path={runtime.platform.path}/tools/
compiler.sdk.path={runtime.platform.path}/tools/sdk
idf.path={runtime.platform.path}
xtensa.path={runtime.tools.xtensa-lx106-elf.path}
compiler.libc.path={runtime.platform.path}/tools/


# ------------------------------------------------------------------------------
# 3. BASE COMPILER FLAGS (Mostly bypassed by CMake)
# ------------------------------------------------------------------------------
compiler.cpreprocessor.flags=-D__ets__ -DICACHE_FLASH -U__STRICT_ANSI__ -D_GNU_SOURCE -DESP8266
compiler.libraries.ldflags=

compiler.c.flags=-std=gnu17 -Os -g -free -fipa-pta -Werror=return-type -Wpointer-arith -Wno-implicit-function-declaration -Wl,-EL -fno-inline-functions -nostdlib -mlongcalls -mtext-section-literals -falign-functions=4 -MMD -ffunction-sections -fdata-sections
compiler.S.flags=-c -g -x assembler-with-cpp -MMD -mlongcalls
compiler.c.elf.flags= -Os -nostdlib -Wl,--no-check-sections  -Wl,-static  "-Tlocal.eagle.flash.ld" -Wl,--gc-sections -Wl,-wrap,system_restart_local -Wl,-wrap,spi_flash_read
compiler.c.elf.libs=-lhal -lphy -lpp  -lwpa -lcrypto -lmain -lwps -lbearssl -lespnow -lsmartconfig -lairkiss  -lm -lc -lgcc
compiler.cpp.flags=-Os -g -free -fipa-pta -Werror=return-type -mlongcalls -mtext-section-literals -fno-rtti -falign -MMD -ffunction-sections
compiler.ar.flags=cru
compiler.elf2hex.flags=
build.extra_flags=

compiler.c.cmd=dummy.py
compiler.S.cmd=dummy.py
compiler.c.elf.cmd=dummy.py
compiler.cpp.cmd=dummy.py
compiler.as.cmd=dummy.py
compiler.ar.cmd=dummy.py
compiler.elf2hex.cmd=dummy.py


# ------------------------------------------------------------------------------
# 4. PRE/POST BUILD HOOKS
# ------------------------------------------------------------------------------
recipe.hooks.prebuild.1.pattern="{python.venv.path}" "{compiler.path}createPreproc.py" -b "{build.path}"

# Linux / macOS
recipe.hooks.postbuild.1.pattern="{python.venv.path}" "{compiler.path}createTemplate.py" -b "{build.path}" -i "{idf.path}" -x "{xtensa.path}" -l "~/Arduino/libraries" -s "{build.source.path}"
recipe.hooks.postbuild.2.pattern="{python.venv.path}" "{compiler.path}set_user_values.py" -b "{build.path}" -c "{build.f_cpu}" -fm "{build.flash_mode}" -ff "{build.flash_freq}" -fs "{build.flash_size_arg}" -l "{build.arduhal_log_level}" -v "{build.variant}" -ps "{build.partition_scheme}" -e "{build.extra_flags}"
recipe.hooks.postbuild.3.pattern="{python.venv.path}" "{compiler.path}compile_linux.py" -b "{build.path}" -i "{idf.path}" -x "{xtensa.path}" -m "{upload.maximum_size}" -f "{build.flash_size_arg}" -C "{cmake.path}" -G "{gcc.path}" -M "{make.path}" -N "{ninja.path}" -P "{python3.path}"

# Windows
recipe.hooks.postbuild.1.pattern.windows="{python.venv.path}" "{compiler.path}createTemplate.py" -b "{build.path}" -i "{idf.path}" -x "{xtensa.path}" -l "~\Documents\Arduino\libraries" -s "{build.source.path}"
recipe.hooks.postbuild.2.pattern.windows="{python.venv.path}" "{compiler.path}set_user_values.py" -b "{build.path}" -c "{build.f_cpu}" -fm "{build.flash_mode}" -ff "{build.flash_freq}" -fs "{build.flash_size_arg}" -l "{build.arduhal_log_level}" -v "{build.variant}" -ps "{build.partition_scheme}" -e "{build.extra_flags}"
recipe.hooks.postbuild.3.pattern.windows="{python.venv.path}" "{compiler.path}compile_windows.py" -b "{build.path}" -i "{idf.path}" -x "{xtensa.path}" -m "{upload.maximum_size}" -f "{build.flash_size_arg}" -C "{cmake.path}" -G "{gcc.path}" -M "{make.path}" -N "{ninja.path}" -P "{python3.path}"


# ------------------------------------------------------------------------------
# 4b. PREPROCESSOR RECIPES
# ------------------------------------------------------------------------------
recipe.preproc.includes="{python.venv.path}" "{compiler.path}dummy.py"
recipe.preproc.macros="{python.venv.path}" "{compiler.path}dummy.py" -o "{preprocessed_file_path}" -c "{source_file}"


# ------------------------------------------------------------------------------
# 5. NATIVE SIZE CALCULATION (Memory Bar)
# ------------------------------------------------------------------------------
recipe.size.pattern="{python.venv.path}" "{compiler.path}dummy.py"
recipe.size.pattern.windows="{python.venv.path}" "{compiler.path}size_windows.py" -b "{build.path}" -i "{idf.path}" -x "{xtensa.path}" -m "{upload.maximum_size}" -f "{build.flash_size_arg}" -C "{cmake.path}" -G "{gcc.path}" -M "{make.path}" -N "{ninja.path}" -P "{python3.path}"
recipe.size.regex=Total image size:\s*(?:~)?\s*([0-9]+)\s+bytes
